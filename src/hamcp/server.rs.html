22/// MCP server for Home Assistant integration.
23#[derive(Debug, Clone)]
24pub struct HaServer {
25    /// HTTP client for Home Assistant API.
26    client: HaClient,
27    /// Tool router for MCP protocol.
28    #[allow(dead_code)]
29    tool_router: ToolRouter<Self>,
30}
31
32impl HaServer {
33    /// Creates a new Home Assistant MCP server.
34    ///
35    /// # Arguments
36    ///
37    /// * `client` - The Home Assistant API client
38    #[must_use]
39    pub fn new(client: HaClient) -> Self {
40        Self {
41            client,
42            tool_router: Self::tool_router(),
43        }
44    }
45}

47/// Serializes a value to pretty-printed JSON.
48///
49/// Returns an error string if serialization fails.
50fn to_json<T: Serialize>(value: &T) -> Result<String, String> {
51    serde_json::to_string_pretty(value).map_err(|e| format!("JSON serialization error: {e}"))
52}

54#[tool_router]
55impl HaServer {
56    /// Checks if the Home Assistant API is running and healthy.
57    #[tool(
58        name = "health_check",
59        description = "Check if the Home Assistant API is running and healthy"
60    )]
61    async fn health_check_tool(&self) -> Result<String, String> {
62        let result = self
63            .client
64            .check_health()
65            .await
66            .map_err(|e| e.to_string())?;
67
68        if result.healthy {
69            Ok(format!("Home Assistant API is healthy: {}", result.message))
70        } else {
71            Err(format!(
72                "Home Assistant API is unhealthy: {}",
73                result.message
74            ))
75        }
76    }
77
78    /// Gets Home Assistant configuration information.
79    #[tool(
80        name = "get_config",
81        description = "Get Home Assistant configuration including location, unit system, and loaded components"
82    )]
83    async fn get_config_tool(&self) -> Result<String, String> {
84        let config = self.client.get_config().await.map_err(|e| e.to_string())?;
85        to_json(&config)
86    }

88    /// Gets all entity states.

92    )]
93    async fn get_states_tool(&self) -> Result<String, String> {
94        let states = self.client.get_states().await.map_err(|e| e.to_string())?;
95        to_json(&states)

98    /// Gets a specific entity's state.
99    #[tool(
100        name = "get_entity",
101        description = "Get the current state of a specific Home Assistant entity by ID"
102    )]
103    async fn get_entity_tool(
104        &self,
105        Parameters(input): Parameters<GetEntityInput>,
106    ) -> Result<String, String> {
107        let state = self
108            .client
109            .get_entity(&input.entity_id)
110            .await
111            .map_err(|e| e.to_string())?;
112        to_json(&state)
113    }
114
115    /// Calls a Home Assistant service.
116    #[tool(
117        name = "call_service",
118        description = "Call a Home Assistant service to control devices. Examples: domain='light', service='turn_on', entity_id='light.living_room'"
119    )]
120    async fn call_service_tool(
121        &self,
122        Parameters(input): Parameters<CallServiceInput>,
123    ) -> Result<String, String> {
124        let response = self
125            .client
126            .call_service(
127                &input.domain,
128                &input.service,
129                input.service_data,
130                input.entity_id.as_deref(),
131                false,
132            )
133            .await
134            .map_err(|e| format!("Failed to call {}.{}: {e}", input.domain, input.service))?;
135        to_json(&response)
136    }
137
138    /// Sets an entity state.
139    #[tool(
140        name = "set_state",
141        description = "Set or update a state for a Home Assistant entity. Creates the entity if it doesn't exist."
142    )]
143    async fn set_state_tool(
144        &self,
145        Parameters(input): Parameters<SetStateInput>,
146    ) -> Result<String, String> {
147        let state_update = crate::models::StateUpdate {
148            state: input.state,

152            .client
153            .set_state(&input.entity_id, &state_update)
154            .await
155            .map_err(|e| e.to_string())?;
156        to_json(&state)
157    }

238    async fn get_history_tool(
239        &self,
240        Parameters(input): Parameters<GetHistoryInput>,

257#[tool_handler]
258impl ServerHandler for HaServer {
259    fn get_info(&self) -> ServerInfo {
260        ServerInfo::new(ServerCapabilities::builder().enable_tools().build())
261            .with_protocol_version(rmcp::model::ProtocolVersion::V_2024_11_05)
262            .with_server_info(Implementation::new("hamcp-rs", env!("CARGO_PKG_VERSION")))
263            .with_instructions(
264                "Home Assistant MCP server for controlling your smart home".to_string(),
265            )
266    }
